###PS_START###
param([string]$Mode)

function Parse-ProxyLine([string]$u) {
    $u = ($u -replace '^\[[^\]]+\]\s*', '').Trim()
    $res = @{ srv = ''; prt = ''; usr = ''; pwd = '' }
    if ($u -match 'server=(?<v>[^&]+)') {
        $res.srv = $matches['v']
        if ($u -match 'port=(?<v>\d+)') { $res.prt = $matches['v'] }
        if ($u -match 'user=(?<v>[^&]+)') { $res.usr = $matches['v'] }
        if ($u -match 'pass=(?<v>[^&]+)') { $res.pwd = $matches['v'] }
    } elseif ($u -match '^(?:socks5?://)?(?:(?<usr>[^:@]+):(?<pwd>[^@]+)@)?(?<srv>[^:/#]+):(?<prt>\d+)(?::(?<usr2>[^:]+):(?<pwd2>.+))?') {
        $res.srv = $matches['srv']
        $res.prt = $matches['prt']
        if ($matches['usr']) { $res.usr = $matches['usr']; $res.pwd = $matches['pwd'] }
        elseif ($matches['usr2']) { $res.usr = $matches['usr2']; $res.pwd = $matches['pwd2'] }
    }
    return $res
}

$dir = (Get-Location).Path

if ($Mode -eq 'check') {
    $goodFile = Join-Path $dir 'chatgpt_good.txt'
    $utf8NoBom = New-Object System.Text.UTF8Encoding $false
    [System.IO.File]::WriteAllText($goodFile, "", $utf8NoBom)

    $lines = Get-Content -Path (Join-Path $dir 'good.txt') -Encoding UTF8 | Where-Object { $_.Trim() -ne '' }
    if ($lines.Count -eq 0) { exit 1 }

    $pool = [runspacefactory]::CreateRunspacePool(150, 150)
    $pool.Open()
    $jobs = New-Object System.Collections.Generic.List[PSCustomObject]

    foreach ($rawLine in $lines) {
        $ps = [powershell]::Create().AddScript({
            param($u)
            $u = $u.Trim()
            $srv = ''; $prt = ''; $usr = ''; $pwd = ''
            if ($u -match 'server=(?<v>[^&]+)') {
                $srv = $matches['v']
                if ($u -match 'port=(?<v>\d+)') { $prt = $matches['v'] }
                if ($u -match 'user=(?<v>[^&]+)') { $usr = $matches['v'] }
                if ($u -match 'pass=(?<v>[^&]+)') { $pwd = $matches['v'] }
            } elseif ($u -match '^(?:socks5?://)?(?:(?<usr>[^:@]+):(?<pwd>[^@]+)@)?(?<srv>[^:/#]+):(?<prt>\d+)(?::(?<usr2>[^:]+):(?<pwd2>.+))?') {
                $srv = $matches['srv']; $prt = $matches['prt']
                if ($matches['usr']) { $usr = $matches['usr']; $pwd = $matches['pwd'] }
                elseif ($matches['usr2']) { $usr = $matches['usr2']; $pwd = $matches['pwd2'] }
            }
            if (-not $srv -or -not $prt) { return $null }

            $sw = [System.Diagnostics.Stopwatch]::StartNew()
            $curlArgs = @('-I', '-s', '-o', 'NUL', '-w', '%{http_code}', '--max-time', '1', '--socks5-hostname', "$srv`:$prt")
            if ($usr -and $pwd) { $curlArgs += @('--proxy-user', "$usr`:$pwd") }
            $curlArgs += 'https://telegram.org'

            $code = & curl.exe @curlArgs
            $sw.Stop()

            if ($LASTEXITCODE -eq 0 -and ($code -eq '200' -or $code -eq '301' -or $code -eq '302' -or $code -eq '307' -or $code -eq '308')) {
                return [PSCustomObject]@{ Ping = [int]$sw.ElapsedMilliseconds; Code = $code; Raw = $u }
            }
            return $null
        }).AddArgument($rawLine)

        $ps.RunspacePool = $pool
        $jobs.Add([PSCustomObject]@{ Pipe = $ps; Handle = $ps.BeginInvoke(); Done = $false })
    }

    $valid = New-Object System.Collections.Generic.List[PSCustomObject]

    try {
        [Console]::TreatControlCAsInput = $true
        while ($true) {
            if ([Console]::KeyAvailable) {
                $key = [Console]::ReadKey($true)
                if (($key.Modifiers -band [ConsoleModifiers]::Control) -and ($key.Key -eq [ConsoleKey]::C)) {
                    Write-Host "`n[СТОП] Проверка прервана! Сортирую по пингу и сохраняю..." -ForegroundColor Yellow
                    break
                }
            }

            $pending = 0
            foreach ($j in $jobs) {
                if (-not $j.Done) {
                    if ($j.Handle.IsCompleted) {
                        $j.Done = $true
                        $res = $j.Pipe.EndInvoke($j.Handle)
                        $j.Pipe.Dispose()
                        if ($res) {
                            $formatted = "[$($res.Ping) ms] $($res.Raw)"
                            Write-Host "[GOOD] ($($res.Ping) ms | HTTP $($res.Code)) -> $($res.Raw)" -ForegroundColor Green
                            $valid.Add([PSCustomObject]@{ Ping = $res.Ping; Line = $formatted })
                            [System.IO.File]::AppendAllText($goodFile, $formatted + [Environment]::NewLine, $utf8NoBom)
                        }
                    } else {
                        $pending++
                    }
                }
            }

            if ($pending -eq 0) { break }
            Start-Sleep -Milliseconds 100
        }
    }
    finally {
        [Console]::TreatControlCAsInput = $false

        if ($valid.Count -gt 0) {
            $sorted = $valid | Sort-Object Ping | ForEach-Object { $_.Line }
            [System.IO.File]::WriteAllLines($goodFile, $sorted, $utf8NoBom)
        }

        Get-Process curl -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
        foreach ($j in $jobs) {
            if (-not $j.Done) {
                try { $j.Pipe.Stop() } catch {}
                try { $j.Pipe.Dispose() } catch {}
            }
        }
        $pool.Close()
        $pool.Dispose()
    }
}
elseif ($Mode -eq 'build') {
    $pData = Parse-ProxyLine $env:PS_TARGET_URL
    $cfg = Get-Content -Path (Join-Path $dir '1.pbprofile') -Raw -Encoding UTF8 | ConvertFrom-Json
    $p = $cfg.ProxyConfigs | Select-Object -First 1
    $p.Host = $pData.srv
    $p.Port = $pData.prt
    $p.Username = $pData.usr
    $p.Password = $pData.pwd
    $p.SendDomainToProxy = $true

    if ($cfg.ProxyRules.Count -eq 0) {
        $cfg.ProxyRules = @(@{ Id = 1; Name = 'All Traffic'; ProcessName = '*'; TargetHosts = '*'; TargetPorts = '*'; Protocol = 'ALL'; Action = 'PROXY'; ProxyId = $p.Id; Enabled = $true })
    }

    $json = $cfg | ConvertTo-Json -Depth 10
    [System.IO.File]::WriteAllText((Join-Path $dir 'active.pbprofile'), $json, (New-Object System.Text.UTF8Encoding $false))
}
